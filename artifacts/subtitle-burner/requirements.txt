# Interview-reframe pipeline (optional — feature is gated to lazy imports
# so the app still boots without these installed). pyannote.audio pulls
# torch (~2 GB) so install only if you want the reframe feature.
#
# Speed knobs (env):
#   DIARIZATION_DEVICE=auto|cuda|mps|cpu   # auto prefers CUDA then MPS
#   DIARIZATION_TORCH_THREADS=8            # Linux CPU parallelism
#   REFRAME_FACE_SAMPLE_FPS=2              # lower = faster face pass
#   DIARIZATION_MODEL=pyannote/speaker-diarization-3.1
#
# IMPORTANT — on minimal Linux sandboxes (Replit, Docker slim), after
# `pip install -r requirements.txt` you must also:
#   pip uninstall -y opencv-contrib-python
#   pip install opencv-python-headless
# mediapipe pulls in `opencv-contrib-python` which dynamic-links against
# libxcb/libGL — those aren't present on most server sandboxes. Headless
# OpenCV has no X11 deps and mediapipe will use it transparently via the
# shared `cv2` module name.
pyannote.audio>=3.1.0
# Pinned below 0.10.22: MediaPipe removed the legacy `mp.solutions` API, and
# this app is written against it (mp.solutions.face_detection for the reframe
# analysis, mp.solutions.selfie_segmentation for the person matte). An
# unpinned install resolves to a release where `mediapipe.solutions` does not
# exist, so both raise AttributeError the first time they run. Verified: 0.10.14
# and 0.10.21 have it; 0.10.35 and 1.2.2 do not.
mediapipe>=0.10.5,<0.10.22
opencv-python-headless>=4.10.0

# Tighter word timestamps. Whisper infers word times from decoder attention and
# they drift; WhisperX aligns them against a phoneme model, and every caption
# highlight and effect in this app is anchored to a word time. On by default
# when installed; set WHISPERX_ALIGN=0 to skip it. Falls back to Whisper's own
# timings whenever the import, the model load or the alignment fails, so a
# missing install costs accuracy rather than breaking transcription.
#
# Deliberately NOT in pyproject.toml: it pulls torch (~2-3 GB), and a deploy
# that builds from pyproject stays lean without it. Install it here when you
# want the accuracy.
whisperx>=3.1.0
